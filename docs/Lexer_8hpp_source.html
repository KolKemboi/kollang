61 std::optional<Token> nextToken() {
62 _skipWhitespace();
63 char c = this->_peek();
64
65 if (c == '\0') {
66 return Token(TokenType::END_OF_FILE, "", this->m_Line, this->m_Column);
67 }
68 if (c == ';') {
69 this->_advance();
70 return Token(TokenType::SEMICOLON, ";", this->m_Line, this->m_Column - 1);
71 }
72 if (c == '(') {
73 this->_advance();
74 return Token(TokenType::LPAREN, "(", this->m_Line, this->m_Column - 1);
75 }
76 if (c == ')') {
77 this->_advance();
78 return Token(TokenType::RPAREN, ")", this->m_Line, this->m_Column - 1);
79 }
80 if (c == '{') {
81 this->_advance();
82 return Token(TokenType::LBRACE, "{", this->m_Line, this->m_Column - 1);
83 }
84 if (c == '}') {
85 this->_advance();
86 return Token(TokenType::RBRACE, "}", this->m_Line, this->m_Column - 1);
87 }
88 if (c == '=') {
89 this->_advance();
90 return Token(TokenType::EQUAL, "=", this->m_Line, this->m_Column - 1);
91 }
92 // if " read until "
93 if (c == '"') {
94 return this->_getString();
95 }
96 if (std::isdigit(c)) {
97 return this->_readNumber();
98 }
99 if (std::isalpha(c) || c == '_') {
100 return this->_readWord();
101 }
102
103 this->_advance();
104 return Token{TokenType::ERROR, std::string_view(&c, 1), m_Line,
105 m_Column - 1}; // silence this thing kwanza
106 }

126 char _peek(size_t offset = 0) const { // just returns what is at pos
127 size_t idx = m_Position + offset;
128 if (idx < m_SourceCode.length())
129 return m_SourceCode[idx];
130 else

134 char _advance() { // decides what to do next
135 char c = this->_peek();
136 if (c == '\n') {
137 this->m_Line++;
138 this->m_Column = 1;
139 } else
140 this->m_Column++;
141 this->m_Position++;
142 return c;
143 }

145 void _skipWhitespace() {
146 while (std::isspace(this->_peek()))
147 this->_advance();

150 Token _readNumber() {
151 size_t start = this->m_Position;
152
153 unsigned int dotCount = 0;
154
155 // for floats, I can add an extra check for a period(.)
156 while (std::isdigit(_peek())) {
157 this->_advance();
158 if (_peek() == '.') {
159 dotCount += 1;
160 _advance();
161 }
162 // raise an error if next is a non numerical
163 }
164 std::string_view num = m_SourceCode.substr(start, m_Position - start);
165 // check if there is a alpha inside num, if, exit and raise
166 if (dotCount == 1) {
167 return Token(TokenType::FLOAT_V, num, m_Line, m_Column - num.length());
168 }
169 return Token(TokenType::INT_V, num, m_Line, m_Column - num.length());
170 }
171
172 Token _getString() {
173 size_t start = m_Position + 1; // move one pos
174 this->_advance(); // next
175 while (_peek() != '"') {
176 this->_advance();
177 }
178 std::string_view word =
179 this->m_SourceCode.substr(start, m_Position - start);
180 _advance();
181 return Token(TokenType::STR_V, word, m_Line, m_Column - word.length());
182 }
183
184 Token _readWord() {
185 size_t start = m_Position;
186 // "m31" => should be valid
187 // "31m" => should be invalid
188 while (std::isalnum(_peek()) || this->_peek() == '_') {
189 this->_advance();
190 }
191
192 std::string_view word =
193 this->m_SourceCode.substr(start, m_Position - start);
194
195 std::unordered_map<std::string_view, TokenType>::const_iterator it =
196 keywords.find(word);
197 if (it != keywords.end()) {
198 return Token(it->second, word, m_Line, m_Column - word.length());
199 }
200 return Token(TokenType::IDENTIFIER, word, m_Line, m_Column - word.length());
201 }
